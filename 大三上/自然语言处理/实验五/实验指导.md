def load_documents(directory_path):
    """
    从指定目录加载文档
    
    参数:
        directory_path: 文档目录路径
    
    返回:
        文档列表
    """
    # 支持txt文件加载
    loader = DirectoryLoader(
        directory_path,
        glob="**/*.txt",
        loader_cls=TextLoader,
        loader_kwargs={'encoding': 'utf-8'}
    )
    documents = loader.load()
    
    print(f"加载了 {len(documents)} 个文档")
    if documents:
        print(f"第一个文档示例: {documents[0].page_content[:100]}...")
    
    return documents

2.2 文档分块
def split_documents(documents, chunk_size=500, chunk_overlap=50):
    """
    将文档分割成小块
    
    参数:
        documents: 文档列表
        chunk_size: 每个块的大小
        chunk_overlap: 块之间的重叠大小
    
    返回:
        文档块列表
    """
    text_splitter = RecursiveCharacterTextSplitter(
        chunk_size=chunk_size,
        chunk_overlap=chunk_overlap,
        length_function=len,
        separators=["\n\n", "\n", "。", "！", "？", "，", " ", ""]
    )
    
    chunks = text_splitter.split_documents(documents)
    
    print(f"文档被分割为 {len(chunks)} 个块")
    if chunks:
        print(f"第一个块示例: {chunks[0].page_content[:100]}...")
    
    return chunks
 
# 分割文档
document_chunks = split_documents(documents)

2.3 文档向量化
def create_vector_store(chunks, embedding_model_name="moka-ai/m3e-base"):
    """
    创建向量存储
    
    参数:
        chunks: 文档块列表
        embedding_model_name: 嵌入模型名称
    
    返回:
        向量存储对象
    """
    # 使用HuggingFace的嵌入模型
    embeddings = HuggingFaceEmbeddings(
        model_name=embedding_model_name,
        model_kwargs={'device': 'cuda' if torch.cuda.is_available() else 'cpu'},
        encode_kwargs={'normalize_embeddings': True}
    )
    
    # 创建FAISS向量存储
    vector_store = FAISS.from_documents(chunks, embeddings)
    
    print(f"向量存储创建完成，包含 {len(chunks)} 个向量")
    
    return vector_store
 
# 创建向量存储

def load_vector_store(index_path="faiss_index", embedding_model_name="moka-ai/m3e-base"):
    """
    加载向量存储
    
    参数:
        index_path: 向量存储路径
        embedding_model_name: 嵌入模型名称
    
    返回:
        向量存储对象
    """
    # 使用HuggingFace的嵌入模型
    embeddings = HuggingFaceEmbeddings(
        model_name=embedding_model_name,
        model_kwargs={'device': 'cuda' if torch.cuda.is_available() else 'cpu'},
        encode_kwargs={'normalize_embeddings': True}
    )
    
    # 加载FAISS向量存储
    vector_store = FAISS.load_local(index_path, embeddings)
    
    print(f"向量存储加载完成")
    
    return vector_store
 
def retrieve_documents(vector_store, query, top_k=3):
    """
    检索相关文档
    
    参数:
        vector_store: 向量存储对象
        query: 查询文本
        top_k: 返回的文档数量
    
    返回:
        相关文档列表及其相似度分数
    """
    # 执行相似度检索
    docs_with_scores = vector_store.similarity_search_with_score(query, k=top_k)
    
    print(f"检索到 {len(docs_with_scores)} 个相关文档")
    
    return docs_with_scores

def load_llm(model_name="THUDM/chatglm3-6b"):
    """
    加载大语言模型
    
    参数:
        model_name: 模型名称
    
    返回:
        tokenizer和model对象
    """
    # 加载tokenizer
    tokenizer = AutoTokenizer.from_pretrained(
        model_name, 
        trust_remote_code=True
    )
    
    # 加载模型
    model = AutoModelForCausalLM.from_pretrained(
        model_name,
        device_map="auto",
        trust_remote_code=True,
        torch_dtype=torch.float16  # 使用半精度浮点数以节省显存
    )
    
    print(f"模型 {model_name} 加载完成")
    
    return tokenizer, model
 
# 加载模型

4.2 实现RAG生成
def format_retrieved_context(docs_with_scores):
    """
    格式化检索到的上下文
    
    参数:
        docs_with_scores: 文档及其相似度分数列表
    
    返回:
        格式化后的上下文字符串
    """
    context = ""
    for i, (doc, score) in enumerate(docs_with_scores):
        context += f"文档 {i+1} (相似度: {score:.4f}):\n{doc.page_content}\n\n"
    
    return context.strip()
 
def generate_response(tokenizer, model, query, context=None, max_length=2048):
    """
    生成回答
    
    参数:
        tokenizer: 分词器
        model: 语言模型
        query: 查询文本
        context: 上下文信息
        max_length: 生成文本的最大长度
    
    返回:
        生成的回答
    """
    if context:
        # RAG模式：包含检索到的上下文
        prompt = f"""请基于以下参考信息回答用户的问题。如果参考信息不足以回答问题，请说明无法回答或提供一般性回答。
 
参考信息：
{context}
 
用户问题：{query}
 
回答："""
    else:
        # 普通模式：不包含上下文
        prompt = f"用户问题：{query}\n\n回答："
    
    # 生成回答
    inputs = tokenizer(prompt, return_tensors="pt").to(model.device)
    outputs = model.generate(
        **inputs,
        max_new_tokens=max_length,
        temperature=0.7,
        top_p=0.9,
        repetition_penalty=1.1
    )
    
    response = tokenizer.decode(outputs[0][inputs.input_ids.shape[1]:], skip_special_tokens=True)
    
    return response

5. 评估RAG系统
def evaluate_responses(query, rag_response, normal_response, reference_answer=None):
    """
    评估RAG和普通生成的回答
    
    参数:
        query: 查询文本
        rag_response: RAG生成的回答
        normal_response: 普通生成的回答
        reference_answer: 参考答案（如果有）
    
    返回:
        评估结果
    """
    # 计算回答长度
    rag_length = len(rag_response)
    normal_length = len(normal_response)
    
    # 计算信息密度（假设标点符号表示信息的分隔）
    import re
    rag_info_points = len(re.findall(r'[。！？；]', rag_response))
    normal_info_points = len(re.findall(r'[。！？；]', normal_response))
    
    # 计算特定实体提及（以中国AI相关实体为例）
    entities = ["百度", "阿里", "腾讯", "华为", "科大讯飞", "商汤", "旷视", "依图", "云从", "清华", "北大"]
    rag_entity_mentions = sum(1 for entity in entities if entity in rag_response)
    normal_entity_mentions = sum(1 for entity in entities if entity in normal_response)
    
    # 输出评估结果
    print("评估结果:")
    print(f"1. 回答长度: RAG={rag_length}, 普通={normal_length}, 差异={(rag_length-normal_length)/normal_length*100:.2f}%")
    print(f"2. 信息点数量: RAG={rag_info_points}, 普通={normal_info_points}, 差异={(rag_info_points-normal_info_points)/normal_info_points*100:.2f}%")
    print(f"3. 实体提及数量: RAG={rag_entity_mentions}, 普通={normal_entity_mentions}")
    
    # 如果有参考答案，可以计算ROUGE或BLEU分数
    if reference_answer:
        from rouge import Rouge
        rouge = Rouge()
        rag_scores = rouge.get_scores(rag_response, reference_answer)[0]
        normal_scores = rouge.get_scores(normal_response, reference_answer)[0]
        
        print("\nROUGE评分:")
        print(f"RAG: ROUGE-1={rag_scores['rouge-1']['f']:.4f}, ROUGE-2={rag_scores['rouge-2']['f']:.4f}, ROUGE-L={rag_scores['rouge-l']['f']:.4f}")
        print(f"普通: ROUGE-1={normal_scores['rouge-1']['f']:.4f}, ROUGE-2={normal_scores['rouge-2']['f']:.4f}, ROUGE-L={normal_scores['rouge-l']['f']:.4f}")
    
    return {
        "length": {"rag": rag_length, "normal": normal_length},
        "info_points": {"rag": rag_info_points, "normal": normal_info_points},
        "entity_mentions": {"rag": rag_entity_mentions, "normal": normal_entity_mentions}
    }
 
# 评估生成的回答
evaluation_results = evaluate_responses(query, rag_response, normal_response)

6.1 改进检索策略
def hybrid_search(vector_store, query, top_k=3, alpha=0.5):
    """
    混合检索策略（结合语义检索和关键词检索）
    
    参数:
        vector_store: 向量存储对象
        query: 查询文本
        top_k: 返回的文档数量
        alpha: 语义检索权重
    
    返回:
        相关文档列表及其相似度分数
    """
    from langchain.retrievers import BM25Retriever
    
    # 语义检索
    semantic_docs = vector_store.similarity_search_with_score(query, k=top_k*2)
    
    # 关键词检索（BM25）
    all_docs = [Document(page_content=doc.page_content, metadata=doc.metadata) for doc, _ in semantic_docs]
    bm25_retriever = BM25Retriever.from_documents(all_docs)
    bm25_retriever.k = top_k*2
    keyword_docs = bm25_retriever.get_relevant_documents(query)
    
    # 合并结果
    doc_scores = {}
    
    # 处理语义检索结果
    for doc, score in semantic_docs:
        doc_id = hash(doc.page_content)
        if doc_id not in doc_scores:
            doc_scores[doc_id] = {"doc": doc, "semantic_score": score, "keyword_score": 0}
        else:
            doc_scores[doc_id]["semantic_score"] = score
    
    # 处理关键词检索结果
    for i, doc in enumerate(keyword_docs):
        doc_id = hash(doc.page_content)
        keyword_score = 1.0 - (i / len(keyword_docs))  # 简化的分数计算
        if doc_id not in doc_scores:
            doc_scores[doc_id] = {"doc": doc, "semantic_score": 0, "keyword_score": keyword_score}
        else:
            doc_scores[doc_id]["keyword_score"] = keyword_score
    
    # 计算混合分数
    for doc_id in doc_scores:
        doc_scores[doc_id]["final_score"] = (
            alpha * doc_scores[doc_id]["semantic_score"] + 
            (1 - alpha) * doc_scores[doc_id]["keyword_score"]
        )
    
    # 排序并返回前top_k个结果
    sorted_docs = sorted(
        doc_scores.values(), 
        key=lambda x: x["final_score"], 
        reverse=True
    )[:top_k]
    
    return [(item["doc"], item["final_score"]) for item in sorted_docs]
 
# 使用混合检索策略

    print(f"文档 {i+1} (混合分数: {score:.4f}):")

def generate_enhanced_response(tokenizer, model, query, docs_with_scores, max_length=2048):
    """
    使用增强提示模板生成回答
    
    参数:
        tokenizer: 分词器
        model: 语言模型
        query: 查询文本
        docs_with_scores: 文档及其相似度分数列表
        max_length: 生成文本的最大长度
    
    返回:
        生成的回答
    """
    # 提取关键信息
    contexts = []
    for i, (doc, score) in enumerate(docs_with_scores):
        contexts.append(f"[文档{i+1}] {doc.page_content}")
    
    context_text = "\n\n".join(contexts)
    
    # 增强提示模板
    prompt = f"""你是一个专业的AI助手，请基于提供的参考文档回答用户的问题。
 
参考文档:
{context_text}
 
用户问题: {query}
 
回答要求:
1. 仅使用参考文档中的信息回答问题
2. 如果参考文档中没有相关信息，请明确说明"根据提供的参考文档，我无法回答这个问题"
3. 回答要全面、准确、客观
4. 回答要有逻辑结构，分点说明
5. 不要在回答中提及"根据参考文档"或"文档中提到"等字样
 
回答:"""
    
    # 生成回答
    inputs = tokenizer(prompt, return_tensors="pt").to(model.device)
    outputs = model.generate(
        **inputs,
        max_new_tokens=max_length,
        temperature=0.5,  # 降低温度以获得更确定性的回答
        top_p=0.9,
        repetition_penalty=1.1
    )
    
    response = tokenizer.decode(outputs[0][inputs.input_ids.shape[1]:], skip_special_tokens=True)
    
    return response
 
# 使用增强提示模板生成回答

def rerank_documents(query, docs_with_scores, model_name="BAAI/bge-reranker-base"):
    """
    对检索到的文档进行重排序
    
    参数:
        query: 查询文本
        docs_with_scores: 文档及其相似度分数列表
        model_name: 重排序模型名称
    
    返回:
        重排序后的文档列表及其相似度分数
    """
    from sentence_transformers import CrossEncoder
    
    # 加载重排序模型
    reranker = CrossEncoder(model_name)
    
    # 准备重排序输入
    pairs = [(query, doc.page_content) for doc, _ in docs_with_scores]
    
    # 计算重排序分数
    scores = reranker.predict(pairs)
    
    # 重新组合文档和分数
    reranked_docs_with_scores = [(docs_with_scores[i][0], float(scores[i])) for i in range(len(scores))]
    
    # 按分数排序
    reranked_docs_with_scores.sort(key=lambda x: x[1], reverse=True)
    
    return reranked_docs_with_scores
