def build_dict_from_file(file_path, encoding='utf-8'):
    """
    从文件构建词典
    
    参数:
        file_path: 词典文件路径
        encoding: 文件编码
    
    返回:
        词典（集合）
    """
    word_dict = set()
    try:
        with open(file_path, 'r', encoding=encoding) as f:
            for line in f:
                word = line.strip().split('\t')[0]  # 假设词典格式为"词\t频率"
                if word:
                    word_dict.add(word)
    except Exception as e:
        print(f"读取词典文件出错: {e}")
        # 创建一个简单的词典作为备用
        word_dict = {"中国", "人民", "共和国", "北京", "天安门", "计算机", "科学", "自然语言", "处理"}
    
    return word_dict

2.2 实现正向最大匹配算法（FMM）
def fmm_segment(text, word_dict, max_word_len=5):
    """
    正向最大匹配算法
    
    参数:
        text: 待分词文本
        word_dict: 词典
        max_word_len: 最大词长
    
    返回:
        分词结果列表
    """
    words = []
    i = 0
    while i < len(text):
        matched = False
        for j in range(max_word_len, 0, -1):
            if i + j > len(text):
                continue
            word = text[i:i+j]
            if word in word_dict:
                words.append(word)
                i += j
                matched = True
                break
        if not matched:  # 未匹配，按字符切分
            words.append(text[i])
            i += 1
    return words

def bmm_segment(text, word_dict, max_word_len=5):
    """
    逆向最大匹配算法
    
    参数:
        text: 待分词文本
        word_dict: 词典
        max_word_len: 最大词长
    
    返回:
        分词结果列表
    """
    words = []
    i = len(text)
    while i > 0:
        matched = False
        for j in range(max_word_len, 0, -1):
            if i - j < 0:
                continue
            word = text[i-j:i]
            if word in word_dict:
                words.insert(0, word)
                i -= j
                matched = True
                break
        if not matched:  # 未匹配，按字符切分
            words.insert(0, text[i-1])
            i -= 1
    return words

def bimm_segment(text, word_dict, max_word_len=5):
    """
    双向最大匹配算法
    
    参数:
        text: 待分词文本
        word_dict: 词典
        max_word_len: 最大词长
    
    返回:
        分词结果列表
    """
    fmm_result = fmm_segment(text, word_dict, max_word_len)
    bmm_result = bmm_segment(text, word_dict, max_word_len)
    
    # 如果正反向分词结果词数不同，则取分词数量较少的那个
    if len(fmm_result) != len(bmm_result):
        return fmm_result if len(fmm_result) < len(bmm_result) else bmm_result
    
    # 如果分词数量相同，则返回单字较少的那个
    fmm_single_word_count = sum(1 for word in fmm_result if len(word) == 1)
    bmm_single_word_count = sum(1 for word in bmm_result if len(word) == 1)
    
    return fmm_result if fmm_single_word_count < bmm_single_word_count else bmm_result

def train_hmm_from_corpus(corpus_file, encoding='utf-8'):
    """
    从语料库训练HMM模型
    
    参数:
        corpus_file: 语料库文件路径
        encoding: 文件编码
    
    返回:
        HMM模型参数: (A, B, pi)
        A: 状态转移概率矩阵
        B: 发射概率矩阵
        pi: 初始状态概率向量
    """
    # 定义状态: B(开始), M(中间), E(结束), S(单字)
    states = ['B', 'M', 'E', 'S']
    
    # 初始化计数器
    state_count = {state: 0 for state in states}
    transition_count = {state: {next_state: 0 for next_state in states} for state in states}
    emission_count = {state: {} for state in states}
    
    try:
        with open(corpus_file, 'r', encoding=encoding) as f:
            for line in f:
                line = line.strip()
                if not line:
                    continue
                
                # 处理已分词的语料
                words = line.split()
                for word in words:
                    if len(word) == 1:
                        state_count['S'] += 1
                        if word not in emission_count['S']:
                            emission_count['S'][word] = 0
                        emission_count['S'][word] += 1
                    else:
                        state_count['B'] += 1
                        if word[0] not in emission_count['B']:
                            emission_count['B'][word[0]] = 0
                        emission_count['B'][word[0]] += 1
                        
                        for i in range(1, len(word) - 1):
                            state_count['M'] += 1
                            if word[i] not in emission_count['M']:
                                emission_count['M'][word[i]] = 0
                            emission_count['M'][word[i]] += 1
                        
                        state_count['E'] += 1
                        if word[-1] not in emission_count['E']:
                            emission_count['E'][word[-1]] = 0
                        emission_count['E'][word[-1]] += 1
                
                # 计算状态转移
                prev_state = None
                for word in words:
                    if len(word) == 1:
                        if prev_state:
                            transition_count[prev_state]['S'] += 1
                        prev_state = 'S'
                    else:
                        if prev_state:
                            transition_count[prev_state]['B'] += 1
                        
                        transition_count['B']['M' if len(word) > 2 else 'E'] += 1
                        
                        for i in range(1, len(word) - 1):
                            transition_count['M']['M' if i < len(word) - 2 else 'E'] += 1
                        
                        prev_state = 'E'
    except Exception as e:
        print(f"训练HMM模型出错: {e}")
        return None
    
    # 计算概率
    A = {state: {next_state: transition_count[state][next_state] / state_count[state] if state_count[state] > 0 else 0 
                for next_state in states} for state in states}
    
    B = {state: {char: emission_count[state][char] / state_count[state] if state_count[state] > 0 else 0 
                for char in emission_count[state]} for state in states}
    
    total_words = sum(state_count.values())
    pi = {state: state_count[state] / total_words if total_words > 0 else 0.25 for state in states}
    
    return A, B, pi
 
def viterbi_segment(text, A, B, pi):
    """
    使用Viterbi算法进行分词
    
    参数:
        text: 待分词文本
        A: 状态转移概率矩阵
        B: 发射概率矩阵
        pi: 初始状态概率向量
    
    返回:
        分词结果列表
    """
    states = ['B', 'M', 'E', 'S']
    
    # 初始化
    V = [{}]
    path = {}
    
    # 初始状态
    for state in states:
        V[0][state] = pi[state] * B[state].get(text[0], 1e-10)
        path[state] = [state]
    
    # 递推
    for t in range(1, len(text)):
        V.append({})
        new_path = {}
        
        for curr_state in states:
            max_prob = -1
            max_state = None
            
            for prev_state in states:
                prob = V[t-1][prev_state] * A[prev_state].get(curr_state, 1e-10) * B[curr_state].get(text[t], 1e-10)
                
                if prob > max_prob:
                    max_prob = prob
                    max_state = prev_state
            
            V[t][curr_state] = max_prob
            new_path[curr_state] = path[max_state] + [curr_state]
        
        path = new_path
    
    # 找出最可能的状态序列
    max_prob = -1
    max_state = None
    
    for state in states:
        if V[len(text)-1][state] > max_prob:
            max_prob = V[len(text)-1][state]
            max_state = state
    
    state_sequence = path[max_state]
    
    # 根据状态序列分词
    words = []
    word = ""
    for i, state in enumerate(state_sequence):
        if state == 'B' or state == 'M':
            word += text[i]
        elif state == 'E':
            word += text[i]
            words.append(word)
            word = ""
        else:  # state == 'S'
            words.append(text[i])
    
    # 处理可能的剩余字符
    if word:
        words.append(word)
    
    return words

def prepare_crf_data(sentences):
    """
    准备CRF训练数据
    
    参数:
        sentences: 分词后的句子列表，每个句子是一个词列表
    
    返回:
        X: 特征列表
        y: 标签列表
    """
    X = []
    y = []
    
    for sentence in sentences:
        chars = []
        labels = []
        
        for word in sentence:
            if len(word) == 1:
                chars.append(word)
                labels.append('S')
            else:
                chars.append(word[0])
                labels.append('B')
                
                for char in word[1:-1]:
                    chars.append(char)
                    labels.append('M')
                
                chars.append(word[-1])
                labels.append('E')
        
        X.append([word2features(chars, i) for i in range(len(chars))])
        y.append(labels)
    
    return X, y
 
def word2features(sent, i):
    """
    提取CRF特征
    
    参数:
        sent: 字符列表
        i: 当前字符索引
    
    返回:
        特征字典
    """
    word = sent[i]
    
    features = {
        'bias': 1.0,
        'word': word,
        'word.isdigit()': word.isdigit(),
        'word.isalpha()': word.isalpha()
    }
    
    if i > 0:
        prev_word = sent[i-1]
        features.update({
            '-1:word': prev_word,
            '-1:word.isdigit()': prev_word.isdigit(),
            '-1:word.isalpha()': prev_word.isalpha(),
            'bigram[-1:]': prev_word + word
        })
    else:
        features['BOS'] = True
    
    if i < len(sent) - 1:
        next_word = sent[i+1]
        features.update({
            '+1:word': next_word,
            '+1:word.isdigit()': next_word.isdigit(),
            '+1:word.isalpha()': next_word.isalpha(),
            'bigram[+1]': word + next_word
        })
    else:
        features['EOS'] = True
    
    return features
 
def crf_segment(text, crf_model):
    """
    使用CRF模型进行分词
    
    参数:
        text: 待分词文本
        crf_model: CRF模型
    
    返回:
        分词结果列表
    """
    chars = list(text)
    features = [word2features(chars, i) for i in range(len(chars))]
    labels = crf_model.predict([features])[0]
    
    words = []
    word = ""
    for i, label in enumerate(labels):
        if label == 'B':
            if word:
                words.append(word)
            word = chars[i]
        elif label == 'M':
            word += chars[i]
        elif label == 'E':
            word += chars[i]
            words.append(word)
            word = ""
        else:  # label == 'S'
            if word:
                words.append(word)
            words.append(chars[i])
            word = ""
    
    if word:
        words.append(word)
    
    return words

def bert_segment(text, model_name="bert-base-chinese"):
    """
    使用BERT进行分词
    
    参数:
        text: 待分词文本
        model_name: BERT模型名称
    
    返回:
        分词结果列表

    tokens = tokenizer.tokenize(text)
    
    # BERT分词会将中文字符单独分开，需要合并
    words = []
    word = ""
    for token in tokens:

        else:
            if word:
                words.append(word)
            word = token
    
    if word:
        words.append(word)
    
    return words

import jieba.posseg as pseg
 
def jieba_ner(text):
    """
    使用jieba进行词性标注和命名实体识别
    
    参数:
        text: 待分析文本
    
    返回:
        词性标注结果列表
    """
    words_with_pos = [(word, pos) for word, pos in pseg.cut(text)]
    
    # 提取命名实体（简单规则）
    entities = []
    for word, pos in words_with_pos:
        if pos.startswith('nr'):  # 人名
            entities.append((word, 'PER'))
        elif pos.startswith('ns'):  # 地名
            entities.append((word, 'LOC'))
        elif pos.startswith('nt'):  # 机构名
            entities.append((word, 'ORG'))
    
    return words_with_pos, entities

def prepare_bert_ner_data(text, tokenizer):
    """
    准备BERT命名实体识别数据
    
    参数:
        text: 待分析文本
        tokenizer: BERT分词器
    
    返回:
        模型输入
    """
    tokens = tokenizer.tokenize(text)
    tokens = ["[CLS]"] + tokens + ["[SEP]"]
    input_ids = tokenizer.convert_tokens_to_ids(tokens)
    attention_mask = [1] * len(input_ids)
    
    return {
        "input_ids": torch.tensor([input_ids]),
        "attention_mask": torch.tensor([attention_mask])
    }
 
def bert_ner(text, model_name="bert-base-chinese"):
    """
    使用BERT进行命名实体识别
    
    参数:
        text: 待分析文本
        model_name: BERT模型名称
    
    返回:
        命名实体识别结果
    """
    # 注意：这里使用的是预训练BERT模型，实际应用中应该使用微调过的NER模型
    tokenizer = BertTokenizer.from_pretrained(model_name)
    model = BertForTokenClassification.from_pretrained(model_name, num_labels=9)  # 假设有9个标签（O, B-PER, I-PER, B-LOC, I-LOC, B-ORG, I-ORG, B-MISC, I-MISC）
    
    inputs = prepare_bert_ner_data(text, tokenizer)
    
    with torch.no_grad():
        outputs = model(**inputs)
        predictions = torch.argmax(outputs.logits, dim=2)
    
    tokens = tokenizer.tokenize(text)
    tokens = ["[CLS]"] + tokens + ["[SEP]"]
    
    # 简化处理：将预测的标签ID映射回标签名称
    # 实际应用中应该使用真实的标签映射
    id2label = {
        0: "O",
        1: "B-PER", 2: "I-PER",
        3: "B-LOC", 4: "I-LOC",
        5: "B-ORG", 6: "I-ORG",
        7: "B-MISC", 8: "I-MISC"
    }
    
    results = []
    for token, prediction in zip(tokens[1:-1], predictions[0][1:-1]):  # 去掉[CLS]和[SEP]
        results.append((token, id2label[prediction.item()]))
    
    return results

def evaluate_segmentation(pred_words, gold_words):
    """
    评估分词效果
    
    参数:
        pred_words: 预测的分词结果
        gold_words: 标准分词结果
    
    返回:
        精确率、召回率、F1值
    """
    # 计算正确分词数
    correct = 0
    for word in pred_words:
        if word in gold_words:
            correct += 1
    
    # 计算精确率、召回率、F1值
    precision = correct / len(pred_words) if pred_words else 0
    recall = correct / len(gold_words) if gold_words else 0
    f1 = 2 * precision * recall / (precision + recall) if precision + recall > 0 else 0
    
    return precision, recall, f1
